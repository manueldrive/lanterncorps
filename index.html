Helper to construct circular arc shapes function createArcPath(innerR, outerR, startAngleDeg, endAngleDeg) { const gap = 0.35; const adjustedStart = startAngleDeg + gap; const adjustedEnd = Math.max(adjustedStart + 0.05, endAngleDeg - gap); const toRad = deg => (deg - 90) * (Math.PI / 180); const a1 = toRad(adjustedStart); const a2 = toRad(adjustedEnd); const x1 = outerR * Math.cos(a1); Math.cos(a1); const y4 = innerR * Math.sin(a1); const largeArc = (adjustedEnd - adjustedStart) > 180 ? 1 : 0; return `M ${x1} ${y1} A ${outerR} ${outerR} 0 ${largeArc} 1 ${x2} ${y2} L ${x3} ${y3} A ${innerR} ${innerR} 0 ${largeArc} 0 ${x4} ${y4} Z`; } // Render Wheel Elements function buildWheel() { booksGroup.innerHTML = ""; bookElements.length = 0; let currentAngle = 0; path.setAttribute("d", d); path.setAttribute("class", "book-block"); const paletteIndex = (startAngle / 360) * (COLOR_PALETTE.length - 1); const baseIndex = Math.floor(paletteIndex); path.setAttribute("fill", COLOR_PALETTE[baseIndex]); path.setAttribute("opacity", "0.9"); // Tooltip path.addEventListener("mousemove", (e) => { tooltip.style.opacity = "1"; tooltip.style.left = ${percentage.toFixed(2)}% (${book.words.toLocaleString()} words)`; }); path.addEventListener("mouseleave", () => { tooltip.style.opacity = "0"; }); booksGroup.appendChild(path); bookElements.push({ element: path, midAngle: (startAngle + endAngle) / 2 }); }); } // GSAP Flying Intro Animation function animateIntro() { bookElements.forEach(({ element, midAngle }) => { const distance = 650 + Math.random() * 350; const spreadAngle = (midAngle + (Math.random() * 40 - 20)) * (Math.PI / 180); const startX = Math.cos(spreadAngle) * distance; const startY = Math.sin(spreadAngle) * distance; const startRotation = Math.random() * 360 - 180; gsap.set(element, { x: startX, y: startY, rotation: startRotation, scale: 0.2, opacity: 0 }); }); gsap.to(".book-block", { x: 0, y: 0, rotation: 0, scale: 1, opacity: 0.9, duration: 1.8, ease: "power3.out", stagger: { amount: 0.9, from: "random" } }); gsap.fromTo(".center-label", { opacity: 0, scale: 0.9 }, { opacity: 1, scale: 1, duration: 1.2, delay: 0.5, ease: "power2.out" } ); } // Initialize on Load window.addEventListener("DOMContentLoaded", ()style => { buildWheel(); animateIntro(); });
